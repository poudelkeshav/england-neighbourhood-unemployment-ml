/* ============================================================
   MERGE ALL 16 VIEWS INTO ONE FINAL ENGLAND LSOA TABLE
   ============================================================ */

DROP TABLE IF EXISTS England_LSOA_Final;

CREATE TABLE England_LSOA_Final AS

SELECT
    /* --------------------------------------------------------
       1. IDENTIFIERS AND TARGET VARIABLE
       -------------------------------------------------------- */
    ea.LSOA_Code,
    ea.LSOA_Name,
    ea.Unemployment_Rate,

    /* --------------------------------------------------------
       2. ECONOMIC ACTIVITY
       -------------------------------------------------------- */
    ea.Student_Rate,
    ea.Retired_Rate,
    ea.Home_Family_Rate,
    ea.LongTerm_Sick_Disabled_Rate,
    ea.Other_Inactive_Rate,

    /* --------------------------------------------------------
       3. AGE
       -------------------------------------------------------- */
    age.Young_Population_Rate,
    age.Working_Age_Rate,
    age.Elderly_Population_Rate,

    /* --------------------------------------------------------
       4. QUALIFICATION
       -------------------------------------------------------- */
    q.No_Qualification_Rate,
    q.Level1_Qualification_Rate,
    q.Level2_Qualification_Rate,
    q.Apprenticeship_Rate,
    q.Level3_Qualification_Rate,
    q.Level4Plus_Qualification_Rate,
    q.Other_Qualification_Rate,
    q.Low_Qualification_Rate,

    /* --------------------------------------------------------
       5. HEALTH
       -------------------------------------------------------- */
    h.Good_Health_Rate,
    h.Fair_Health_Rate,
    h.Poor_Health_Rate,

    /* --------------------------------------------------------
       6. DISABILITY
       -------------------------------------------------------- */
    d.Disability_Rate,
    d.Not_Disabled_Rate,

    /* --------------------------------------------------------
       7. MIGRATION
       -------------------------------------------------------- */
    m.Migrant_Within_UK_Rate,
    m.Migrant_Outside_UK_Rate,

    /* --------------------------------------------------------
       8. ETHNICITY
       -------------------------------------------------------- */
    eth.White_Rate,
    eth.Asian_Rate,
    eth.Black_Rate,
    eth.Mixed_Rate,
    eth.Other_Ethnic_Group_Rate,

    /* --------------------------------------------------------
       9. CAR OR VAN AVAILABILITY
       -------------------------------------------------------- */
    car.No_Car_Van_Rate,
    car.One_Car_Van_Rate,
    car.Two_Cars_Vans_Rate,
    car.ThreePlus_Cars_Vans_Rate,
    car.Any_Car_Van_Rate,

    /* --------------------------------------------------------
       10. TRAVEL TO WORK
       -------------------------------------------------------- */
    tr.Work_From_Home_Rate,
    tr.Public_Transport_Rate,
    tr.Car_Commute_Rate,
    tr.Active_Travel_Rate,
    tr.Other_Transport_Rate,

    /* --------------------------------------------------------
       11. TENURE
       Corrected complete tenure categories
       -------------------------------------------------------- */
    ten.Owned_Rate,
    ten.Shared_Ownership_Rate,
    ten.Social_Rented_Rate,
    ten.Private_Rented_Rate,
    ten.Lives_Rent_Free_Rate,

    /* --------------------------------------------------------
       12. ACCOMMODATION
       -------------------------------------------------------- */
    acc.House_Rate,
    acc.Purpose_Built_Flat_Rate,
    acc.Converted_Building_Rate,
    acc.Other_Temporary_Accommodation_Rate,

    /* --------------------------------------------------------
       13. NS-SEC
       -------------------------------------------------------- */
    ns.Managerial_Rate,
    ns.Intermediate_Occupations_Rate,
    ns.Small_Employers_Rate,
    ns.Lower_Occupations_Rate,
    ns.Never_Worked_Rate,
    ns.Full_Time_Students_Rate,

    /* --------------------------------------------------------
       14. INDUSTRY
       -------------------------------------------------------- */
    ind.Primary_Industry_Rate,
    ind.Production_Construction_Rate,
    ind.Consumer_Services_Rate,
    ind.Business_Knowledge_Services_Rate,
    ind.Public_Services_Rate,
    ind.Other_Services_Rate,

    /* --------------------------------------------------------
       15. POPULATION DENSITY
       -------------------------------------------------------- */
    pd.Population_per_km2,

    /* --------------------------------------------------------
       16. RURAL / URBAN CLASSIFICATION
       -------------------------------------------------------- */
    ru.RUC21CD,
    ru.RUC21NM,
    ru.Urban_Rural_Flag,

    /* --------------------------------------------------------
       17. INDEX OF MULTIPLE DEPRIVATION
       -------------------------------------------------------- */
    imd.IMD_Score,
    imd.IMD_Rank,
    imd.IMD_Decile,
    imd.Income_Deprivation_Score,
    imd.Employment_Deprivation_Score,
    imd.Education_Deprivation_Score,
    imd.Health_Deprivation_Score,
    imd.Crime_Deprivation_Score,
    imd.Housing_Services_Deprivation_Score,
    imd.Living_Environment_Deprivation_Score

FROM vw_economic_activity AS ea

LEFT JOIN vw_age AS age
    ON ea.LSOA_Code = age.LSOA_Code

LEFT JOIN vw_qualification AS q
    ON ea.LSOA_Code = q.LSOA_Code

LEFT JOIN vw_health AS h
    ON ea.LSOA_Code = h.LSOA_Code

LEFT JOIN vw_disability AS d
    ON ea.LSOA_Code = d.LSOA_Code

LEFT JOIN vw_migration AS m
    ON ea.LSOA_Code = m.LSOA_Code

LEFT JOIN vw_ethnicity AS eth
    ON ea.LSOA_Code = eth.LSOA_Code

LEFT JOIN vw_car_availability AS car
    ON ea.LSOA_Code = car.LSOA_Code

LEFT JOIN vw_travel AS tr
    ON ea.LSOA_Code = tr.LSOA_Code

LEFT JOIN vw_tenure AS ten
    ON ea.LSOA_Code = ten.LSOA_Code

LEFT JOIN vw_accommodation AS acc
    ON ea.LSOA_Code = acc.LSOA_Code

LEFT JOIN vw_nssec AS ns
    ON ea.LSOA_Code = ns.LSOA_Code

LEFT JOIN vw_industry AS ind
    ON ea.LSOA_Code = ind.LSOA_Code

LEFT JOIN vw_population_density AS pd
    ON ea.LSOA_Code = pd.LSOA_Code

LEFT JOIN vw_rural_urban AS ru
    ON ea.LSOA_Code = ru.LSOA_Code

LEFT JOIN vw_imd AS imd
    ON ea.LSOA_Code = imd.LSOA_Code

WHERE ea.LSOA_Code LIKE 'E%';


/* ============================================================
   FINAL TABLE VALIDATION
   ============================================================ */


/* 1. Display the first 10 records */

SELECT *
FROM England_LSOA_Final
LIMIT 10;


/* 2. Count final rows */

SELECT
    COUNT(*) AS Total_Rows
FROM England_LSOA_Final;


/* 3. Count unique LSOAs */

SELECT
    COUNT(DISTINCT LSOA_Code) AS Unique_LSOAs
FROM England_LSOA_Final;


/* 4. Check duplicate LSOA codes */

SELECT
    LSOA_Code,
    COUNT(*) AS Record_Count
FROM England_LSOA_Final
GROUP BY LSOA_Code
HAVING COUNT(*) > 1;


/* 5. Check missing LSOA identifiers */

SELECT
    COUNT(*) AS Missing_LSOA_Codes
FROM England_LSOA_Final
WHERE LSOA_Code IS NULL;


/* 6. Check whether any joined dataset failed to match */

SELECT
    SUM(CASE WHEN age.LSOA_Code IS NULL THEN 1 ELSE 0 END)
        AS Missing_Age,

    SUM(CASE WHEN q.LSOA_Code IS NULL THEN 1 ELSE 0 END)
        AS Missing_Qualification,

    SUM(CASE WHEN h.LSOA_Code IS NULL THEN 1 ELSE 0 END)
        AS Missing_Health,

    SUM(CASE WHEN d.LSOA_Code IS NULL THEN 1 ELSE 0 END)
        AS Missing_Disability,

    SUM(CASE WHEN m.LSOA_Code IS NULL THEN 1 ELSE 0 END)
        AS Missing_Migration,

    SUM(CASE WHEN eth.LSOA_Code IS NULL THEN 1 ELSE 0 END)
        AS Missing_Ethnicity,

    SUM(CASE WHEN car.LSOA_Code IS NULL THEN 1 ELSE 0 END)
        AS Missing_Car_Availability,

    SUM(CASE WHEN tr.LSOA_Code IS NULL THEN 1 ELSE 0 END)
        AS Missing_Travel,

    SUM(CASE WHEN ten.LSOA_Code IS NULL THEN 1 ELSE 0 END)
        AS Missing_Tenure,

    SUM(CASE WHEN acc.LSOA_Code IS NULL THEN 1 ELSE 0 END)
        AS Missing_Accommodation,

    SUM(CASE WHEN ns.LSOA_Code IS NULL THEN 1 ELSE 0 END)
        AS Missing_NSSEC,

    SUM(CASE WHEN ind.LSOA_Code IS NULL THEN 1 ELSE 0 END)
        AS Missing_Industry,

    SUM(CASE WHEN pd.LSOA_Code IS NULL THEN 1 ELSE 0 END)
        AS Missing_Population_Density,

    SUM(CASE WHEN ru.LSOA_Code IS NULL THEN 1 ELSE 0 END)
        AS Missing_Rural_Urban,

    SUM(CASE WHEN imd.LSOA_Code IS NULL THEN 1 ELSE 0 END)
        AS Missing_IMD

FROM vw_economic_activity AS ea

LEFT JOIN vw_age AS age
    ON ea.LSOA_Code = age.LSOA_Code

LEFT JOIN vw_qualification AS q
    ON ea.LSOA_Code = q.LSOA_Code

LEFT JOIN vw_health AS h
    ON ea.LSOA_Code = h.LSOA_Code

LEFT JOIN vw_disability AS d
    ON ea.LSOA_Code = d.LSOA_Code

LEFT JOIN vw_migration AS m
    ON ea.LSOA_Code = m.LSOA_Code

LEFT JOIN vw_ethnicity AS eth
    ON ea.LSOA_Code = eth.LSOA_Code

LEFT JOIN vw_car_availability AS car
    ON ea.LSOA_Code = car.LSOA_Code

LEFT JOIN vw_travel AS tr
    ON ea.LSOA_Code = tr.LSOA_Code

LEFT JOIN vw_tenure AS ten
    ON ea.LSOA_Code = ten.LSOA_Code

LEFT JOIN vw_accommodation AS acc
    ON ea.LSOA_Code = acc.LSOA_Code

LEFT JOIN vw_nssec AS ns
    ON ea.LSOA_Code = ns.LSOA_Code

LEFT JOIN vw_industry AS ind
    ON ea.LSOA_Code = ind.LSOA_Code

LEFT JOIN vw_population_density AS pd
    ON ea.LSOA_Code = pd.LSOA_Code

LEFT JOIN vw_rural_urban AS ru
    ON ea.LSOA_Code = ru.LSOA_Code

LEFT JOIN vw_imd AS imd
    ON ea.LSOA_Code = imd.LSOA_Code

WHERE ea.LSOA_Code LIKE 'E%';


/* 7. Check the number of columns in the final table */

SELECT
    COUNT(*) AS Total_Columns
FROM pragma_table_info('England_LSOA_Final');


/* 8. Display the final table structure */

PRAGMA table_info(England_LSOA_Final);


/* 9. Final summary */

SELECT
    COUNT(*) AS Total_Rows,
    COUNT(DISTINCT LSOA_Code) AS Unique_LSOAs,
    COUNT(*) - COUNT(DISTINCT LSOA_Code) AS Duplicate_Rows,
    MIN(Unemployment_Rate) AS Minimum_Unemployment_Rate,
    MAX(Unemployment_Rate) AS Maximum_Unemployment_Rate,
    ROUND(AVG(Unemployment_Rate), 4)
        AS Average_Unemployment_Rate
FROM England_LSOA_Final;
